# -----------------------------
# HYPOTHESIS TESTING (t-test)
# -----------------------------

# Sample dataset (50 observations)

set.seed(123)

sample_data <- rnorm(
  50,
  mean = 52,
  sd = 10
)

# Perform one-sample t-test

t_test <- t.test(sample_data, mu = 50)

print(t_test)


# -----------------------------
# GRAPH (Visualization)
# -----------------------------

hist(

  probability = TRUE,
  main = "Hypothesis Testing (t-test)",
  col = "lightblue",
  xlab = "Values"
)
